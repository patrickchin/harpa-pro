import GuideStep from "../../components/docs/GuideStep.astro";

<GuideStep number={1} title="Enter your email">
  Open Harpa Pro and enter the email address you use for the app. Tap **Send
  Code**.
</GuideStep>

<GuideStep number={2} title="Verify the code">
  Enter the one-time code from the Harpa Pro email. Verifying a first-time
  email creates the account; there is no separate password or registration
  form.
</GuideStep>

<GuideStep
  number={3}
  title="Start the first project"
  screenshot="projects-list"
  screenshotAlt="Projects screen with Add project ready for a new Harpa Pro account"
>
  Tap **Add project**, enter its details, and create it. Open **Reports**, tap
  **New report**, and begin a draft.
</GuideStep>

Next, follow [Generate an AI report](/docs/guides/generate-ai-report) or learn
how to [create and manage projects](/docs/guides/manage-projects).
