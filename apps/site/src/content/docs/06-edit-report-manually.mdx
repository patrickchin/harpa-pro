import GuideStep from "../../components/docs/GuideStep.astro";

<GuideStep number={1} title="Open a draft">
  Open the report and select **Report**. Manual edits are made on individual
  cards; there is no separate Edit tab.
</GuideStep>

<GuideStep
  number={2}
  title="Edit the affected card"
  screenshot="final-report-sections"
  screenshotAlt="Generated report cards with pencil actions for focused edits"
>
  Find the weather, worker, material, issue, action, or section that needs a
  correction. Tap its pencil action, update the fields, and tap **Save**.
</GuideStep>

<GuideStep number={3} title="Review or unfinalize first">
  Wait for saving to finish and review the card in context. If the report is
  finalized, an Owner or Editor must use **Report Actions** → **Unfinalize
  report** before editing, then finalize it again when ready.
</GuideStep>
