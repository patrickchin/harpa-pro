import GuideStep from "../../components/docs/GuideStep.astro";

<GuideStep
  number={1}
  title="Start a report"
  screenshot="reports-list"
  screenshotAlt="Harpa Pro Reports list with the New report button"
>
  Open a project, tap **Reports**, then tap **New report**. Harpa Pro creates a
  draft and opens its note workspace.
</GuideStep>

<GuideStep
  number={2}
  title="Capture what happened"
  screenshot="voice-recording"
  screenshotAlt="Voice note being recorded in a Harpa Pro draft report"
>
  Add short text notes, photos, or voice notes. Send each update into the
  timeline, and wait for uploads or voice processing to finish.
</GuideStep>

<GuideStep number={3} title="Generate the report">
  Tap **Generate report** after at least one note is ready. Keep the draft open
  while generation completes, then open **Report**.
</GuideStep>

<GuideStep
  number={4}
  title="Review and correct it"
  screenshot="final-report-issues"
  screenshotAlt="Generated report cards ready for review in Harpa Pro"
>
  Check every report card. Tap a card's pencil action to correct it and save.
  There is no separate Edit tab. If source notes change, Harpa Pro queues an
  updated draft while the report remains open.
</GuideStep>

<GuideStep number={5} title="Finalize">
  Verify names, dates, quantities, issues, and commitments, then tap **Finalize
  report**. A finalized report is read-only and no longer updates when notes
  change.
</GuideStep>

> Generated output is a draft. Review it against the site record before
> finalizing or sharing it.
